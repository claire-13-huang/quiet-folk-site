const duration=value=>{const n=Math.round(value||0);return n>=60?`${Math.floor(n/60)}m ${n%60}s`:`${n}s`};

function chart(data){const hourly=data.range==='24H',step=hourly?3600000:86400000,count=(hourly?24:data.range==='7D'?7:30)+1,end=Math.floor(data.now/step),points=Array.from({length:count},(_,i)=>({bucket:end-count+1+i,count:0}));for(const row of data.chart){const p=points.find(p=>p.bucket===row.bucket);if(p)p.count=row.count}const max=Math.max(3,...points.map(p=>p.count)),w=900,h=215,left=35,right=12,top=10,bottom=32;const x=i=>left+i*(w-left-right)/(count-1),y=n=>h-bottom-n*(h-top-bottom)/max;const line=points.map((p,i)=>`${i?'L':'M'}${x(i)},${y(p.count)}`).join(' ');let svg=`<svg viewBox="0 0 ${w} ${h}" role="img" aria-label="${esc(data.range)} visits; ${number(data.metrics.visits)} total"><defs><linearGradient id="wash" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="#c4a56d" stop-opacity=".14"/><stop offset="100%" stop-color="#c4a56d" stop-opacity="0"/></linearGradient></defs>`;for(let i=0;i<4;i++){const n=Math.round(max*i/3),py=y(n);svg+=`<line x1="${left}" x2="${w-right}" y1="${py}" y2="${py}" stroke="#36322e" stroke-dasharray="3 5"/><text x="0" y="${py+4}">${n}</text>`}svg+=`<path d="${line} L${x(count-1)},${y(0)} L${left},${y(0)} Z" fill="url(#wash)"/><path d="${line}" fill="none" stroke="#c4a56d" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"/>`;points.forEach((p,i)=>{svg+=`<circle cx="${x(i)}" cy="${y(p.count)}" r="3" fill="#c4a56d"><title>${esc(date(p.bucket*step))}: ${p.count} visits</title></circle>`;if(i===0||i===count-1||i%Math.ceil(count/5)===0){const label=new Date(p.bucket*step).toLocaleString(undefined,hourly?{hour:'2-digit'}:{month:'short',day:'numeric'});svg+=`<text x="${x(i)}" y="${h-5}" text-anchor="${i===0?'start':i===count-1?'end':'middle'}">${esc(label)}</text>`}});$('chart').innerHTML=svg+'</svg>'}
